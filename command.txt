cat > ~/.cop.sh <<'EOF'
cop() {
    if [ $# -eq 0 ]; then
        echo "Usage: cop <a|h|c|j|name> [more...]"
        return 1
    fi

    local tmp input pattern f
    tmp=$(mktemp) || return 1

    for input in "$@"; do
        case "$input" in
            a) pattern="*" ;;
            h) pattern="*.html" ;;
            c) pattern="*.css" ;;
            j) pattern="*.js" ;;
            *) pattern="${input}*" ;;
        esac

        find . -type f -name "$pattern" \
            -not -path '*/node_modules/*' \
            -not -path '*/.git/*' | sort | while IFS= read -r f; do
            grep -Iq . "$f" 2>/dev/null || continue
            printf '\n===== %s =====\n' "$f"
            cat "$f"
        done >> "$tmp"
    done

    if [ ! -s "$tmp" ]; then
        echo "✗ No files found: $*"
        rm -f "$tmp"
        return 1
    fi

    xclip -selection clipboard < "$tmp"
    rm -f "$tmp"
    echo "✓ Copied: $*"
}
EOF
